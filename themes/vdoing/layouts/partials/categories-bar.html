"length" 10 "category" "") }} length 为 "all" 时显示全部分类 */ -}} {{- $page := . -}} {{- $length := "all" -}} {{- $active := "" -}} {{- if reflect.IsMap . -}} {{- $page = .page -}} {{- with .length }}{{ $length = . }}{{ end -}} {{- with .category }}{{ $active = . }}{{ end -}} {{- end -}} {{- $posts := slice -}} {{- range site.RegularPages -}} {{- if ne .Params.article false }}{{ $posts = $posts | append . }}{{ end -}} {{- end -}} {{- $counts := dict -}} {{- range $posts -}} {{- range .Params.categories -}} {{- with . -}} {{- $n := index $counts . | default 0 -}} {{- $counts = merge $counts (dict . (add $n 1)) -}} {{- end -}} {{- end -}} {{- end -}} {{- $items := slice -}} {{- range $k, $v := $counts -}} {{- $items = $items | append (dict "key" (printf "%06d|%s" $v $k) "name" $k "count" $v) -}} {{- end -}} {{- $items = sort $items "key" "desc" -}} {{- $shown := $items -}} {{- if ne (printf "%v" $length) "all" }}{{ $shown = first (int $length) $items }}{{ end -}}
